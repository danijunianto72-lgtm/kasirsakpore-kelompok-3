CREATE TABLE barang_new (
    kodebarang   INTEGER PRIMARY KEY,
    skubarang    VARCHAR(50) NOT NULL,
    nama         VARCHAR(100) NOT NULL,
    hargabarang  NUMERIC(12,2) NOT NULL,
    stok         INTEGER,
    kategori     VARCHAR(100),
    gambar       VARCHAR(255),
    hargapokok   NUMERIC(12,2),
    ppn          NUMERIC(12,2),
    satuan       VARCHAR(200)
);

-- Pindahkan data
INSERT INTO barang_new (kodebarang, skubarang, nama, hargabarang, stok, kategori, gambar, hargapokok, ppn, satuan)
SELECT kodebarang, skubarang, nama, hargabarang, stok, kategori, gambar, hargapokok, ppn, satuan
FROM barang;

-- Hapus tabel lama (opsional)
DROP TABLE barang;

-- Ganti nama tabel baru
ALTER TABLE barang_new RENAME TO barang;
